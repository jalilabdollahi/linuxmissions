  sed -E 's#OLD#NEW#g' FILE     -E: extended regex; any character can replace /
                                as the delimiter; g: every match on the line
  \.                            a literal dot
  ([0-9]{2})                    a group: exactly two digits
  \1 \2 \3                      in the replacement: the text of group 1, 2, 3
  -e 'cmd1' -e 'cmd2'           several commands in one sed run
  sed -i                        edit files in place (GNU). On macOS: sed -i '' ...
                                (BSD sed takes the word after -i as a backup suffix)
  find templates -name '*.html' -exec sed ... {} +     every template, subfolders included
