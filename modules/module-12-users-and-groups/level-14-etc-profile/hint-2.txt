  env -i LM_ETC=$SANDBOX/etc sh -c '. etc/profile; printenv http_proxy https_proxy JAVA_HOME'
  env -i LM_ETC=$SANDBOX/etc bash --login --noprofile -c '. etc/profile; printenv JAVA_HOME'
