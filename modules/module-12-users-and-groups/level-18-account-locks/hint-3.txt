grep '^  - ' TICKET.md | while IFS= read -r l; do
  name=${l#  - }; name=${name%% <*}; name=${name%% (email:*}
  email=$(grep -o '[a-z.-]*@corp\.example' <<< "$l")
  user=$(awk -F: -v e="$email" -v n="$name" '{ split($5, g, ","); if (g[1] == n || g[4] == e) print $1 }' etc/passwd)
  echo "$name -> $user"
  bin/usermod -L -e 1 "$user"
done

- grep '^  - ' TICKET.md   one line per leaver on the ticket
- name=... email=...      cut the full name and the address out of that line
- awk ... split($5, g, ",")   matches the whole first part (name) or the fourth part (email) of the comment; deploy matches neither
- bin/usermod -L -e 1     locks the password and sets the expiry to day 1 (long past)
