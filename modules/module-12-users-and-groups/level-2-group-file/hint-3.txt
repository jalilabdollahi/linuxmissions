g=$(awk -F: '$1 == "docker" { print $3 }' etc/group)
{ awk -F: '$1 == "docker" { print $4 }' etc/group | tr ',' '\n'; awk -F: -v g="$g" '$4 == g { print $1 }' etc/passwd; } | grep -v '^$' | sort -u > docker-members.txt
cat docker-members.txt

- g=$(...)            docker's GID, read from field 3 of its own line ($1 == "docker" skips docker-ci)
- print $4 | tr       the supplementary members, one per line
- $4 == g in passwd   the accounts whose PRIMARY group is docker; they are never in the member list
- grep -v '^$'        removes the empty entry the trailing comma produces
- sort -u             one name per line, no duplicates
