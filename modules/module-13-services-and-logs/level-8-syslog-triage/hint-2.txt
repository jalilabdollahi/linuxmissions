  $3 >= "02:00:00" && $3 <= "02:59:59"        times compare correctly as strings here
  prog = $5; sub(/\[[0-9]+\]/, "", prog); sub(/:$/, "", prog)    tag -> program name
  "message repeated 412 times: [ ... ]"       is $6 $7 $8 ...: add $8 instead of 1
  count[prog] += n  ...  END { for (p in count) print count[p], p }
