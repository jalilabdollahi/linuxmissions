cat REQUEST.txt
truncate -s $(grep -o 'exactly [0-9]*' REQUEST.txt | grep -o '[0-9]*')M logs.img
mke2fs -q -F -t ext4 -L $(grep -o 'label: *[A-Z]*' REQUEST.txt | grep -o '[A-Z]*$') -m $(grep -o 'reserved: *[0-9]*' REQUEST.txt | grep -o '[0-9]*$') -i $(grep -o 'every [0-9]* bytes' REQUEST.txt | grep -o '[0-9]*') logs.img
dumpe2fs -h logs.img | grep -E 'volume name|Block count|Reserved block count|Inode count'

- cat REQUEST.txt    read the numbers (the $(grep ...) parts copy them for you; you can type them yourself)
- truncate -s ...M   create the image with exactly the requested size
- -t ext4 -F -q      ext4 type, no question for a regular file, quiet output
- -L -m -i           label, reserved percent, one inode per that many bytes
- dumpe2fs -h        read the superblock to check the result
