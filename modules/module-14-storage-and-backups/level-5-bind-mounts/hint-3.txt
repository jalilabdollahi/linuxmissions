awk '{ for (i = 7; $i != "-"; i++); fs = $(i + 1); if (fs !~ /^(ext[234]|xfs)$/) next; if ($4 == "/") main[$3] = $5; else { n++; bmp[n] = $5; bdev[n] = $3; broot[n] = $4 } } END { for (k = 1; k <= n; k++) { m = main[bdev[k]]; if (m == "/") m = ""; print bmp[k], m broot[k] } }' mountinfo.txt | sort > binds.txt
awk '{ print $1 "/" }' binds.txt > backup-excludes.txt
cat binds.txt backup-excludes.txt

- for (i = 7; $i != "-"; i++)   find the "-" separator, because the optional fields make the column of the file system type move
- fs !~ /^(ext[234]|xfs)$/      skip proc, sysfs, tmpfs and the like
- $4 == "/"                     a normal mount: remember where its device (major:minor) lives
- else {...}                    root is not "/": a bind of a subdirectory, keep it
- m broot[k]                    normal mount point + root = the source directory (m is empty for "/")
- print $1 "/"                  anchored directory pattern for rsync
