sed -i '/[[:space:]]\/tmp[[:space:]]/d' etc/fstab
echo "tmpfs  /tmp  tmpfs  size=$(grep -o 'Size limit: [0-9]*[GM]' TICKET.txt | grep -o '[0-9]*[GM]'),mode=1777,nosuid,nodev  0  0" >> etc/fstab
df -Pk /dev/shm | awk 'NR == 2 { print $2 }' > shm.txt
cat etc/fstab shm.txt

- sed -i '/ \/tmp /d'   delete the old disk line for /tmp
- echo ... >> etc/fstab append the tmpfs line (the $(grep ...) copies the size from the ticket; you can type it)
- size, mode, nosuid, nodev   the options the ticket lists, and no noexec
- df -Pk /dev/shm       the 1K-block size of /dev/shm; awk keeps the second line's second field
