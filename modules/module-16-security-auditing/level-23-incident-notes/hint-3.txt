fp=$(grep "Accepted publickey.* from $ip " var/log/auth.log | awk '{print $NF}' | tail -1)
printf '%s\n%s\n' "$ip" "$fp" > iocs.txt
while read -r line; do [ "$(echo "$line" | ssh-keygen -lf - | awk '{print $2}')" = "$fp" ] || echo "$line"; done < home/deploy/.ssh/authorized_keys > ak.new && cat ak.new > home/deploy/.ssh/authorized_keys && rm ak.new
crontab -l | grep -v 'base64 -d' | crontab -
sed -i '/\.sysd\/sw/d' home/deploy/.bashrc
rm -f home/deploy/.cache/.sysd/sh

- crontab         the beacon is the only job that pipes a base64 blob into a shell; the others stay (write back through the mock crontab)
