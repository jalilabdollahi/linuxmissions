- for ... sleep 0.1        take 30 samples, 0.1 s apart
- ps -eo ppid=,pcpu=,args= parent PID, CPU share and command line of every process, no header
- awk '$2 > 30 && ...'     keep only CPU-heavy short-lived step processes, print their parent PID
- sort | uniq -c | sort -rn  which parent shows up most among them
- awk 'NR==1 {print $2}'   the top parent's PID goes into culprit.txt
