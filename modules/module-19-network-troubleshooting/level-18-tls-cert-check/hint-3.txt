Type this loop at the prompt (all lines up to "done > tls-report.tsv"):

grep -v '^#' services.txt | while read -r svc host; do
  c=certs/$svc/fullchain.pem
  if ! openssl x509 -in "$c" -noout -checkend 0 >/dev/null; then st=EXPIRED
  elif ! openssl verify -CAfile certs/internal-root-ca.pem -untrusted "$c" "$c" >/dev/null 2>&1; then st=CHAIN
  elif ! openssl x509 -in "$c" -noout -checkhost "$host" | grep -q 'does match'; then st=NAME
  else st=OK
  fi
  printf '%s\t%s\n' "$svc" "$st"
done > tls-report.tsv

- -checkend 0           fails when notAfter is in the past
- verify -untrusted     builds the chain from what the server sends; fails when the intermediate isn't there
- -checkhost "$host"    compares the hostname with the subjectAltName list (wildcards included), like a browser
- printf '%s\t%s\n'     "service<TAB>status"
