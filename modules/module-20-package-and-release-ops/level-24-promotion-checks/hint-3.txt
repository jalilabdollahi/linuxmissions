cat > promote.sh <<'SCRIPT'
#!/bin/bash
here=$(cd "$(dirname "$0")" && pwd)
b=${1:?usage: promote.sh BUILD_DIR}
reject() { echo "REJECT: $*"; exit 1; }
gpgv --keyring "$here/keys/release.gpg" "$b/SHA256SUMS.sig" "$b/SHA256SUMS" >/dev/null 2>&1 || reject signature
(cd "$b" && sha256sum -c --status SHA256SUMS) >/dev/null 2>&1 || reject checksum
art=$(ls "$b"/agent-*.tar.gz 2>/dev/null | head -n 1)
ver=$(basename "$art" .tar.gz); ver=${ver#agent-}
dpkg --compare-versions "$ver" gt "$(cat "$here/prod/VERSION")" || reject version
re=$(printf '%s' "$ver" | sed 's/\./\\./g')
grep -qE "^## $re - " "$b/CHANGELOG.md" || reject changelog
list=$(tar --numeric-owner -tvzf "$art" 2>/dev/null) || reject layout
printf '%s\n' "$list" | awk -v top="agent-$ver" '{ perm = $1; own = $2; name = $NF } index(name, top "/") != 1 && name != top { bad = 1 } own != "0/0" { bad = 1 } name ~ /(^|\/)\.DS_Store$/ || name ~ /\.o$/ || name ~ /~$/ { bad = 1 } { rel = name; sub("^" top "/?", "", rel) } perm ~ /^d/ { if (perm != "drwxr-xr-x") bad = 1; next } perm ~ /^-/ { want = "-rw-r--r--"; if (rel ~ /^bin\//) want = "-rwxr-xr-x"; else if (rel ~ /^conf\//) want = "-rw-r-----"; if (perm != want) bad = 1; next } { bad = 1 } END { exit bad }' || reject layout
echo PROMOTE
SCRIPT
chmod +x promote.sh
for b in builds/*; do echo "$b: $(./promote.sh "$b")"; done

- reject()            prints the reason line and stops with status 1
- gpgv                valid signature by the release key, else "signature"
- sha256sum -c --status   all files match SHA256SUMS, else "checksum"
- dpkg --compare-versions ... gt   newer than prod in Debian order (1.10.0 > 1.9.5), else "version"
- grep with escaped dots   the exact heading "## <version> - " (no near misses), else "changelog"
- tar -tv | awk       every member under agent-<version>/, owner 0/0, no junk, modes by directory, else "layout"
