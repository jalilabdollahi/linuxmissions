Type this at the prompt. It rewrites parse_artifact.sh with built-ins only:

cat > parse_artifact.sh <<'EOF'
#!/bin/bash
# parse_artifact.sh FILE - NAME-VERSION-OS-ARCH.EXT, bash built-ins only
file=${1##*/}
re='^([a-z0-9][a-z0-9-]*)-([0-9]+\.[0-9]+\.[0-9]+)-([a-z0-9]+)-([a-z0-9]+)\.(tar\.gz|tar\.xz|zip)$'
if [[ $file =~ $re ]]; then
  echo "name=${BASH_REMATCH[1]} version=${BASH_REMATCH[2]} os=${BASH_REMATCH[3]} arch=${BASH_REMATCH[4]} ext=${BASH_REMATCH[5]}"
else
  echo "parse_artifact: not an artifact name: $1" >&2
  exit 1
fi
EOF

- ${1##*/}                 the file name without its directory, no basename needed
- re='^(...)-(...)-...$'   the whole format, anchored; each ( ) captures one part
- ([0-9]+\.[0-9]+\.[0-9]+)  the version is what pins down where the name ends
- ${BASH_REMATCH[1]} ...   the captured parts, in order
- >&2; exit 1              anything else is rejected

Check: env -i PATH=/nonexistent /bin/bash ./parse_artifact.sh dist/k8s-node-agent-0.19.0-linux-arm64.tar.xz
