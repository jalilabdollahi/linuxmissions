Type this at the prompt. It rewrites upload_batch.sh around an array:

cat > upload_batch.sh <<'EOF'
#!/bin/bash
# upload_batch.sh MANIFEST - upload every file listed in MANIFEST in batches of 4.
mapfile -t lines < "$1"
files=()
for line in "${lines[@]}"; do
  [ -n "$line" ] && files+=("$line")
done

for ((i = 0; i < ${#files[@]}; i += 4)); do
  ./tools/upload "${files[@]:i:4}"
done
echo "uploaded ${#files[@]} files"
EOF

- mapfile -t lines < "$1"          one array element per manifest line (bash 4+)
- [ -n "$line" ] && files+=(...)   keep only the non-empty lines
- "${files[@]:i:4}"                the next (up to) 4 names, each one still a single argument
- ${#files[@]}                     how many names there are

Check: rm -f upload.log; ./upload_batch.sh manifest.txt; cat upload.log
