cat > detect.sh <<'SCRIPT'
#!/bin/bash
echo "host,date,value,zscore"
awk -F, 'NR > 1 && $3 ~ /^-?[0-9]+(\.[0-9]+)?$/ { n[$2]++; s[$2] += $3; v[$2, n[$2]] = $3; d[$2, n[$2]] = $1 }
END { for (h in n) { m = s[h] / n[h]; q = 0; for (i = 1; i <= n[h]; i++) q += (v[h, i] - m) ^ 2; sd = sqrt(q / n[h]); if (sd == 0) continue
      for (i = 1; i <= n[h]; i++) { z = (v[h, i] - m) / sd; if (z > 3 || z < -3) printf "%s,%s,%s,%.2f\n", h, d[h, i], v[h, i], z } } }' "$1" | sort -t, -k1,1 -k2,2
SCRIPT
chmod +x detect.sh
./detect.sh requests.csv > anomalies.csv
cat anomalies.csv

- NR > 1 && $3 ~ /.../       skip the header and every value that is not a number
- n, s, v[h, i], d[h, i]     per host: count, sum, and each value with its date
- END: m, q, sd              each host's own mean and population standard deviation
- if (sd == 0) continue      a constant host has nothing to flag (and no division by 0)
- z > 3 || z < -3            spikes and drops
- "$1" | sort ...            read the file given as argument; sort by host, then date
