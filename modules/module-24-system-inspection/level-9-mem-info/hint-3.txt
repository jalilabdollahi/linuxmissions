echo "host,available_mib,consumer" > memory.csv
for f in capture/*.meminfo; do awk -v h=$(basename $f .meminfo) '/^MemAvailable:/ { a = int($2/1024) } /^HugePages_Total:/ { ht = $2 } /^Hugepagesize:/ { hs = $2 } /^Shmem:/ { sh = $2 } /^SUnreclaim:/ { su = $2 } END { c = "HugePages"; m = ht*hs; if (sh > m) { c = "Shmem"; m = sh } if (su > m) { c = "SUnreclaim" } print h "," a "," c }' $f; done >> memory.csv
cat memory.csv

- /^MemAvailable:/ { a = int($2/1024) }   kB to whole MiB, rounded down
- ht*hs                                    huge pages in kB: HugePages_Total times Hugepagesize
- /^Shmem:/ and /^SUnreclaim:/             the other two consumers, already in kB
- the END block                            picks the largest of the three and prints the CSV row
- for f in capture/*.meminfo               one row per host, in name order
