  b=$(grep -o 'boundary="[^"]*"' bundle.eml | cut -d'"' -f2); mkdir -p parts
      read the boundary string from the header
  csplit -z -s -f parts/p- -b '%02d' bundle.eml "/^--$b/" '{*}'
      p-00 holds the mail headers, p-01 to p-03 the three parts, p-04 the closing boundary
  for f in parts/p-0[1-3]; do sed '1,/^$/d' "$f" | base64 -d > "$f.bin"; file -b "$f.bin" | cut -c1-60; done
      strip each part's headers, decode the body, and let file name what it really is
  kernel=$(zcat parts/p-0[1-3].bin 2>/dev/null | grep -o 'Linux version [^ ]*' | cut -d' ' -f3)
      zcat only succeeds on the gzip part; the other two are skipped by 2>/dev/null
  for f in parts/p-0[1-3].bin; do case $(file -b "$f") in *UTF-16*) errors=$(iconv -f UTF-16 -t UTF-8 "$f" | grep -c ' ERROR ');; *ASCII*|*UTF-8*) setting=$(grep 'Recommended setting:' "$f" | sed 's/.*: //');; esac; done
      the UTF-16 part is converted before counting ERROR lines; the text part gives the recommended setting
  printf 'errors: %s\nkernel: %s\nsetting: %s\n' "$errors" "$kernel" "$setting" > summary.txt; cat summary.txt
