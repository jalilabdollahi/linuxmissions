M=$(awk '$1=="cleanup" {split($5, a, ":"); print a[2]}' SCHEDULE.tsv)
sed -i "s|^\* 2 \* \* \*|$M 2 * * *|" crontab.txt
sed -i 's|date +%F|date +\\%F|' crontab.txt
chmod +x jobs/export.sh
sed -i "/^MAILTO/a PATH=$PWD/usr/local/bin:/usr/bin:/bin" crontab.txt
sed -i "s|^\([0-9,]* 2 \* \* \* \)\($PWD/jobs/backup.sh\)|\1flock -n $PWD/backup.lock \2|; /backup.sh/s|>/dev/null\$|>/dev/null \|\| true|" crontab.txt
./cron-sim

- first sed     the cleanup line fires at its minute instead of every minute of hour 2
- second sed    \% so cron does not cut the command at the percent sign
- chmod +x      the export script can be executed
- PATH line     cron can find collect-stats in usr/local/bin
- last sed      the backups take a lock without waiting, and a skipped start exits 0
